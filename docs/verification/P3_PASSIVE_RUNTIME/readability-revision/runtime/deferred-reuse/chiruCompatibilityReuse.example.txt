import test from 'node:test';import {createHash} from 'node:crypto';import assert from 'node:assert/strict';import {mkdtempSync,readFileSync,writeFileSync,mkdirSync,copyFileSync,rmSync} from 'node:fs';import {tmpdir} from 'node:os';import path from 'node:path';import {verifyChiruBrowserReuse} from '../scripts/reuseChiruBrowser.mjs';
const proofPath='docs/verification/P3_PASSIVE_RUNTIME/readability-revision/runtime/compatibility/reuse-proof.json';
test('new current-source compatibility reports reuse only exact verified sources and reject changed source or a forged result',()=>{
 const proof=JSON.parse(readFileSync(proofPath));assert.equal(verifyChiruBrowserReuse('special').status,'PASS');assert.equal(verifyChiruBrowserReuse('awakening').status,'PASS');const root=mkdtempSync(path.join(tmpdir(),'exact-compat-'));
 try{for(const p of new Set([proofPath,...Object.keys(proof.groups.special.sourceSha256),...Object.keys(proof.groups.special.reportSha256),...Object.keys(proof.groups.special.renderSha256)])){mkdirSync(path.dirname(path.join(root,p)),{recursive:true});copyFileSync(p,path.join(root,p));}assert.equal(verifyChiruBrowserReuse('special',root).status,'PASS');const source='src/runtime/assetPresenter.js',original=readFileSync(path.join(root,source));writeFileSync(path.join(root,source),'changed');assert.throws(()=>verifyChiruBrowserReuse('special',root));writeFileSync(path.join(root,source),original);const report=Object.keys(proof.groups.special.reportSha256)[0];writeFileSync(path.join(root,report),'{}');assert.throws(()=>verifyChiruBrowserReuse('special',root));proof.groups.special.reportSha256[report]=createHash('sha256').update('{}').digest('hex');writeFileSync(path.join(root,proofPath),JSON.stringify(proof));assert.throws(()=>verifyChiruBrowserReuse('special',root));}finally{rmSync(root,{recursive:true,force:true});}
});
